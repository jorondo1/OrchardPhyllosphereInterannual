==== model: Radagast the Grower ====

$likelihood
Dv ~ dlnorm(mu, sigma[Mg])

$main_model
mu <- loga[Mg] + gamma * (Mo - 1) + yr_eff + b_deg * deg_h_z + 
    b_precip * precip_72h_z + b_seq * seq_depth_z

$gamma_def
gamma <- s_conv + gap_shift * (Mg - 1)

$prior_loga
loga[Mg] ~ dnorm(2, 2)

$prior_s
s_conv ~ dnorm(0, 1)

$prior_gs
gap_shift ~ dnorm(0, 1)

$pr_sigma
sigma[Mg] ~ dhalfnorm(0, 1)

$yr_eff_def
yr_eff <- yr1 * (Yr == 1) + yr2 * (Yr == 2) - (yr1 + yr2) * (Yr == 
    3)

$prior_yr1
yr1 ~ dnorm(0, 1)

$prior_yr2
yr2 ~ dnorm(0, 1)

$prior_deg
b_deg ~ dnorm(0, 1)

$prior_precip
b_precip ~ dnorm(0, 1)

$prior_seq
b_seq ~ dnorm(0, 1)
